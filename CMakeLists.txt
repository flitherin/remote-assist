cmake_minimum_required(VERSION 3.21)
project(RemoteAssistScaffold VERSION 0.1 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_library(assist_session STATIC src/session.c)
target_include_directories(assist_session PUBLIC include)
if(WIN32)
  enable_language(RC)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 is required")
  endif()
  foreach(role IN ITEMS client server)
    add_executable(assist_${role} WIN32 src/window.c src/chat.c src/screen.c resources/app.rc)
    target_include_directories(assist_${role} PRIVATE resources)
    target_compile_definitions(assist_${role} PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(assist_${role} PRIVATE assist_session user32 gdi32 comctl32 ws2_32)
    if(MSVC)
      target_compile_options(assist_${role} PRIVATE /utf-8 /W4)
      target_link_options(assist_${role} PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT)
    elseif(MINGW)
      target_compile_options(assist_${role} PRIVATE -Wall -Wextra -Werror)
      target_link_options(assist_${role} PRIVATE -municode -static -static-libgcc -Wl,--dynamicbase,--nxcompat)
    endif()
  endforeach()
  target_compile_definitions(assist_server PRIVATE ASSIST_HOST=1)
else()
  message(STATUS "Only the portable session library is built on this platform")
endif()
